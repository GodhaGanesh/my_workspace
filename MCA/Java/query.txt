SELECT ENAME, JOB, DNAME AS DEPARTMENT
FROM EMP,DEPT
WHERE EMP.DEPTNO = DEPT.DEPTNO;

SELECT E1.ENAME AS EMPLOYEE, E2.ENAME AS MANAGER
FROM EMP E1, EMP E2
WHERE E1.MGR = E2.EMPNO;

SELECT ENAME, NVL((SAL+COMM), SAL) AS INCOME
FROM EMP;

CREATE TABLE TRAINS(
    TNO NUMBER(5) PRIMARY KEY,
    SRC VARCHAR2(15) DEFAULT 'VSKP' NOT NULL,
    DESTINATION VARCHAR2(15) NOT NULL,
    TNAME VARCHAR2(20),
    DTIME DATE,
    ATIME DATE,
    STATUS NUMBER(1),
    CHECK (STATUS IN (0,1)),
    CHECK (TNO BETWEEN 10000 AND 99999)
);

INSERT INTO TRAINS 
VALUES(&TNO,'&SRC','&DESTINATION','&TNAME',TO_DATE('&DTIME','DD-MM-YYYY HH24:MI'),TO_DATE('&ATIME','DD-MM-YYYY HH24:MI'),&STATUS);
